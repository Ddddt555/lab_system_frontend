<template>
  <div class="login-container">
    <div class="login-box">
      <h2>实验教学管理系统</h2>

      <!-- 身份选择 -->
      <el-radio-group v-model="role" class="input-item">
        <el-radio value="student">学生登录</el-radio>
        <el-radio value="teacher">教师登录</el-radio>
        <el-radio value="admin">管理员登录</el-radio>
      </el-radio-group>

      <!-- 账号输入框 -->

        show-password

          size="large"

          @keyup.enter="handleLogin"

        <img
          :src="captchaImage"
          @click="fetchCaptcha"
          class="captcha-img"
          title="点击刷新验证码"
          alt="验证码"
        />
      </div>

      <!-- 登录按钮 -->
      <el-button
        type="primary"
        size="large"

        @click="handleLogin"

      >

      </el-button>

<script setup>
import { ref, onMounted } from 'vue'
import { ElMessage } from 'element-plus'
import { useRouter } from 'vue-router'
import request from '../utils/request'

const role = ref('student')
const username = ref('')
const password = ref('')
const loading = ref(false)
const router = useRouter()

const captchaKey = ref('')
const captchaCode = ref('')
const captchaImage = ref('')

const fetchCaptcha = async () => {
  try {
    const res = await request.get('/captcha')
    if (res.code === 200) {
      captchaKey.value = res.data.captchaKey
      captchaImage.value = res.data.captchaImage
      captchaCode.value = ''
    }
  } catch (error) {
    ElMessage.error('获取验证码失败，请检查网络！')
  }
}

onMounted(() => {
  fetchCaptcha()
})

const handleLogin = async () => {
  if (!username.value || !password.value) {
    ElMessage.warning('请输入账号和密码！')
    return
  }
  if (!captchaCode.value) {
    ElMessage.warning('请输入验证码！')
    return
  }

  loading.value = true
  try {
    let apiUrl = ''
    if (role.value === 'student') {
      apiUrl = '/login/student'
    } else if (role.value === 'teacher') {
      apiUrl = '/login/teacher'
    } else {
      apiUrl = '/login/admin'
    }

    const res = await request.post(apiUrl, null, {
      params: {
        username: username.value,
        password: password.value,
        captchaKey: captchaKey.value,
        captchaCode: captchaCode.value
      }
    })

    if (res.code === 200) {
      // ⚠️ [第一步改造] user 现在是 { userNo, name, role } 三个字段
      //    role 由后端返回，前端不再手动赋值
      const userData = res.data.user
      const token = res.data.token

      ElMessage.success('登录成功！欢迎你，' + userData.name)

      localStorage.setItem('loginUser', JSON.stringify(userData))
      localStorage.setItem('token', token)

      // ⚠️ [第一步改造] 按后端返回的 role 跳转
      if (userData.role === 'student') {
        router.push('/student-home')
      } else if (userData.role === 'teacher') {
        router.push('/teacher-home')
      } else {
        router.push('/admin-home')
      }
    } else {
      ElMessage.error(res.msg)
      fetchCaptcha()
    }
  } catch (error) {
    console.error(error)
    if (!error.response || error.response.status !== 401) {
      ElMessage.error('网络错误，请检查后端是否启动！')
    }
    fetchCaptcha()
  } finally {
    loading.value = false
  }
}
</script>

<script setup>
import { ref, onMounted, computed, nextTick, onBeforeUnmount } from 'vue'
import { useRouter } from 'vue-router'
import request from '../utils/request'
import { ElMessage, ElMessageBox } from 'element-plus'
import * as echarts from 'echarts'
import { Bell } from '@element-plus/icons-vue'
import { WsClient } from '../utils/ws'

const router = useRouter()
const loginUser = JSON.parse(localStorage.getItem('loginUser'))
const workNo = loginUser ? loginUser.userNo : ''
if (!workNo) router.push('/')

const screenWidth = ref(window.innerWidth)
const isMobile = computed(() => screenWidth.value < 768)
const handleResize = () => { screenWidth.value = window.innerWidth }

const activeTab = ref('sessions')
const loading = ref(false)

// 表格数据（分页）
const sessionList = ref([])
const projectList = ref([])
const scoreList = ref([])
const totalSessions = ref(0)
const totalProjects = ref(0)
const totalScores = ref(0)

// 下拉框数据（全量）
const myClassOptions = ref([])
const labOptions = ref([])
const semesterOptions = ref([])
const allProjects = ref([])

const sessionPage = ref(1)
const projectPage = ref(1)
const scorePage = ref(1)
const pageSize = ref(10)

const selectedClassNo = ref('')
const bookingDialogVisible = ref(false)
const studentList = ref([])
const reportDialogVisible = ref(false)
const reportList = ref([])

const projectDialog = ref(false)
const newProject = ref({ projectName: '', classNo: '', hours: 2, standard: '' })

const sessionDialog = ref(false)
const newSession = ref({ projectNo: '', roomNo: '', semester: '', weekNum: 1, dayOfWeek: 1, period: '1-2节', capacity: 50 })
const selectedLabCapacity = ref(100)

const previewDialogVisible = ref(false)
const previewFileUrl = ref('')
const previewFileName = ref('')

const pwdDialog = ref(false)
const pwdForm = ref({ oldPassword: '', newPassword: '', confirmPassword: '' })

const publishDialogVisible = ref(false)
const noticeForm = ref({ title: '', content: '' })
const noticeDrawerVisible = ref(false)
const noticeList = ref([])
const hasUnread = ref(false)

// 签到相关变量
const attendanceDialogVisible = ref(false)
const attendanceCode = ref('')
const attendanceList = ref([])
const currentSessionNo = ref('')
let wsClient = null 

// ⚠️ 权重设置相关变量
const weightDialogVisible = ref(false)
const weightForm = ref({ dailyWeight: 30, reportWeight: 70 })
const currentDailyWeight = ref(30)
const currentReportWeight = ref(70)

const scoreChartRef = ref(null)
const rateChartRef = ref(null)
let scoreChart = null
let rateChart = null

onMounted(() => {
  fetchSessions(); fetchProjects(); fetchMyClasses(); fetchLabs(); fetchSemesters(); fetchNotices();
  fetchAllProjectsForDropdown();
  window.addEventListener('resize', handleResize)
})

onBeforeUnmount(() => {
  closeWebSocket() 
  window.removeEventListener('resize', handleResize)
  if (scoreChart) scoreChart.dispose()
  if (rateChart) rateChart.dispose()
  if (previewFileUrl.value && previewFileUrl.value.startsWith('blob:')) {
    window.URL.revokeObjectURL(previewFileUrl.value)
  }
})

const handleTabChange = (name) => {
  if (name === 'stats') nextTick(() => renderCharts())
}

// ========== 表格分页请求 ==========
const fetchSessions = async () => {
  loading.value = true
  try {
    const res = await request.get('/teacher/my-sessions', { params: { page: sessionPage.value, size: pageSize.value } })
    if (res.code === 200) { sessionList.value = res.data.records; totalSessions.value = res.data.total; }
  } finally {
    loading.value = false
  }
}

const fetchProjects = async () => { 
  try { 
    const res = await request.get('/teacher/project/list', { params: { page: projectPage.value, size: pageSize.value } }); 
    if (res.code === 200) { projectList.value = res.data.records; totalProjects.value = res.data.total; } 
  } catch (e) {} 
}

const fetchCourseScores = async () => { 
  if (!selectedClassNo.value) return; 
  loading.value = true; 
  try { 
    const res = await request.get('/teacher/scores', { params: { classNo: selectedClassNo.value, page: scorePage.value, size: pageSize.value } }); 
    if (res.code === 200) { scoreList.value = res.data.records; totalScores.value = res.data.total; } 
  } catch (e) { 
    ElMessage.error('获取成绩单失败') 
  } finally { 
    loading.value = false 
  } 
}

// 页码切换事件
const handleSessionPageChange = (val) => { sessionPage.value = val; fetchSessions(); }
const handleProjectPageChange = (val) => { projectPage.value = val; fetchProjects(); }
const handleScorePageChange = (val) => { scorePage.value = val; fetchCourseScores(); }

// ⚠️ 选课下拉框切换时，同步更新权重变量
const handleClassChange = () => { 
  scorePage.value = 1; 
  const targetClass = myClassOptions.value.find(c => c.classNo === selectedClassNo.value);
  if (targetClass) {
    currentDailyWeight.value = targetClass.dailyWeight || 30;
    currentReportWeight.value = targetClass.reportWeight || 70;
  } else {
    currentDailyWeight.value = 30;
    currentReportWeight.value = 70;
  }
  fetchCourseScores(); 
}

// ========== 下拉框全量请求 ==========
const fetchAllProjectsForDropdown = async () => {
  try {
    const res = await request.get('/teacher/project/list');
    if (res.code === 200) allProjects.value = res.data;
  } catch (e) {}
}

const fetchMyClasses = async () => { 
  try { 
    const res = await request.get('/teacher/my-classes'); 
    if (res.code === 200) {
      // ⚠️ 这里假设后端返回了 dailyWeight 和 reportWeight
      myClassOptions.value = res.data.map(item => ({
        ...item,
        dailyWeight: item.dailyWeight || 30,
        reportWeight: item.reportWeight || 70
      }))
    } 
  } catch (e) {} 
}

const fetchLabs = async () => { try { const res = await request.get('/teacher/labs'); if (res.code === 200) labOptions.value = res.data } catch (e) {} }
const fetchSemesters = async () => {
  try {
    const res = await request.get('/teacher/semesters')
    if (res.code === 200) {
      semesterOptions.value = res.data
      if (semesterOptions.value.length > 0 && !newSession.value.semester) newSession.value.semester = semesterOptions.value[0].semesterName
    }
  } catch (e) {}
}

const fetchNotices = async () => {
  try {
    const res = await request.get('/notice/list')
    if (res.code === 200) {
      noticeList.value = res.data
      if (noticeList.value.length > 0) {
        const latestTime = new Date(noticeList.value[0].createTime).getTime()
        const lastReadStr = localStorage.getItem('lastReadNoticeTime')
        const lastReadTime = lastReadStr ? new Date(lastReadStr).getTime() : 0
        hasUnread.value = latestTime > lastReadTime
      }
    }
  } catch (e) {}
}

const openNoticeDrawer = () => { noticeDrawerVisible.value = true; hasUnread.value = false; if (noticeList.value.length > 0) localStorage.setItem('lastReadNoticeTime', noticeList.value[0].createTime) }
const openPublishDialog = () => { noticeForm.value = { title: '', content: '' }; publishDialogVisible.value = true }

const submitPublish = async () => {
  if (!noticeForm.value.title || !noticeForm.value.content) return ElMessage.warning('标题和内容不能为空')
  try {
    const res = await request.post('/notice/publish', null, {
      params: { title: noticeForm.value.title, content: noticeForm.value.content, targetRole: 'student' }
    })
    if (res.code === 200) { ElMessage.success('通知发布成功！'); publishDialogVisible.value = false; fetchNotices() }
    else { ElMessage.error(res.msg) }
  } catch (e) { ElMessage.error('发布失败') }
}

const renderCharts = async () => {
  if (scoreChartRef.value) {
    if (!scoreChart) scoreChart = echarts.init(scoreChartRef.value)
    const res = await request.get('/teacher/stats/score-distribution')
    if (res.code === 200) {
      const data = res.data
      scoreChart.setOption({
        tooltip: { trigger: 'item' }, legend: { top: 'bottom' },
        series: [{ name: '成绩分布', type: 'pie', radius: ['40%', '70%'], avoidLabelOverlap: false, itemStyle: { borderRadius: 10, borderColor: '#fff', borderWidth: 2 }, label: { show: false, position: 'center' }, emphasis: { label: { show: true, fontSize: 20, fontWeight: 'bold' } }, labelLine: { show: false },
            data: [{ value: data['优秀'], name: '优秀 (90-100)', itemStyle: { color: '#67C23A' } }, { value: data['良好'], name: '良好 (80-89)', itemStyle: { color: '#409EFF' } }, { value: data['中等'], name: '中等 (70-79)', itemStyle: { color: '#E6A23C' } }, { value: data['及格'], name: '及格 (60-69)', itemStyle: { color: '#909399' } }, { value: data['不及格'], name: '不及格 (<60)', itemStyle: { color: '#F56C6C' } }]
        }]
      }, true)
    }
  }
  if (rateChartRef.value) {
    if (!rateChart) rateChart = echarts.init(rateChartRef.value)
    const res = await request.get('/teacher/stats/experiment-rate')
    if (res.code === 200) {
      const data = res.data

const initWebSocket = async (sessionNo) => {

    const res = await request.get('/ws-ticket')
    if (res.code !== 200) return ElMessage.error('获取WS连接凭证失败')

    const ticket = res.data
    const wsUrl = `${import.meta.env.VITE_WS_BASE_URL}/ws/attendance/${sessionNo}?ticket=${ticket}`
    
    wsClient = new WsClient(wsUrl)
    
    wsClient.onMessageCallback = (message) => {
      if (message === 'NEW_SIGN_IN') {
        fetchAttendanceList()
        ElMessage.success('有学生签到啦！')
      }
    }
    
    wsClient.connect()
    
  } catch (e) {
    console.error('WebSocket 初始化失败', e)
  }
}

const closeWebSocket = () => {
  if (wsClient) {
    wsClient.close()
    wsClient = null
  }
}

// ⚠️ 初始化全局WS
const initGlobalWebSocket = async () => {
  try {
    const res = await request.get('/ws-ticket')
    if (res.code !== 200) return
     const wsUrl = `${import.meta.env.VITE_WS_BASE_URL}/ws/global?ticket=${res.data}`
    
    globalWs = new WsClient(wsUrl)
    globalWs.onMessageCallback = (msg) => {
      try {
        const data = JSON.parse(msg)
        if (data.type === 'NEW_NOTICE') {
          fetchNotices() 
          ElMessage({ type: 'warning', message: '收到一条新系统通知！' })
        }
      } catch (e) {}
    }
    globalWs.connect()
  } catch (e) {
    console.error('全局 WS 建立失败', e)
  }
}

const downloadTemplate = () => {
  request.get('/admin/student/import-template', {
    responseType: 'blob'
  }).then(res => {
    const blob = new Blob([res], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' })
    const url = window.URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = url
    link.setAttribute('download', '学生导入模板.xlsx')
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
    window.URL.revokeObjectURL(url)
    ElMessage.success('模板下载成功！')
  }).catch(e => {
    console.error(e)
    ElMessage.error('下载模板失败，请检查后端服务')
  })
}

const handleImportExcel = async (file) => {
  const rawFile = file.raw
  if (!rawFile) return
  const formData = new FormData()
  formData.append('file', rawFile)
  try {
    const res = await request.post('/admin/student/import', formData, { 
      headers: { 'Content-Type': 'multipart/form-data' } 
    })
    if (res.code === 200) {
      ElMessage.success(res.data)
      fetchStudents() 
      fetchAllDropdownData() // 导入成功后刷新下拉框
    } else {
      ElMessage.error(res.msg)
    }
  } catch (e) {
    ElMessage.error('导入失败，请检查文件格式！')
  }
}

const openPwdDialog = () => { pwdForm.value = { oldPassword: '', newPassword: '', confirmPassword: '' }; pwdDialog.value = true }
const submitPwd = async () => {
  const { oldPassword, newPassword, confirmPassword } = pwdForm.value
  if (!oldPassword || !newPassword || !confirmPassword) return ElMessage.warning('请填写完整信息！')
  if (newPassword !== confirmPassword) return ElMessage.warning('两次输入的新密码不一致！')
  const res = await request.post('/admin/update-password', null, { params: { oldPassword, newPassword } })
  if (res.code === 200) { 
    ElMessage.success('修改成功，请重新登录！'); 
    pwdDialog.value = false;

  const token = localStorage.getItem('token')
  const isLogin = !!loginUserStr && !!token

  // 1. 准备前往登录页

      if (user.role === 'student') return next('/student-home')

      if (user.role === 'admin') return next('/admin-home')
    }

    return next()
  }

  // 2. 如果去其他页面，但没登录，强行踢回登录页
  if (!isLogin) {
    return next('/')
  }

  // 3. 已经登录，校验角色权限（防篡改）
  const user = JSON.parse(loginUserStr)
  const requiredRole = to.meta.role // 获取目标路由要求的角色

  // 如果目标路由需要角色要求，并且当前用户的角色不匹配
  if (requiredRole && user.role !== requiredRole) {

    if (user.role === 'admin') return next('/admin-home')

  // 校验通过，正常放行
  next()

const request = axios.create({
  // ⚠️ 核心修改：使用环境变量。开发环境自动读取 .env.development，生产环境读取 .env.production
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',

// 请求拦截器：每次发请求前，自动带上 Token
request.interceptors.request.use(
  config => {
    const token = localStorage.getItem('token')
    if (token) {
      config.headers['Authorization'] = 'Bearer ' + token
    }
    return config
  },
  error => {
    return Promise.reject(error)
  }
)

// 响应拦截器：处理 Token 过期、Blob 下载异常等情况
request.interceptors.response.use(
  response => {
    // ⚠️ 优化：如果是文件下载请求（blob），直接返回整个 response，不要返回 response.data
    // 否则前端无法获取到 header 里的文件名（Content-Disposition）
    if (response.config.responseType === 'blob') {
      return response;
    }
    
    // 正常 JSON 数据，直接返回后端的 Result 对象
    return response.data 
  },
  error => {
    if (error.response) {
      const status = error.response.status
      
      if (status === 401) {
        // 401 代表 Token 过期、无效或被拉黑
        ElMessage.error('登录已过期或失效，请重新登录！')
        localStorage.removeItem('loginUser')
        localStorage.removeItem('token')
        // 避免重复跳转
        if (router.currentRoute.value.path !== '/') {
          router.push('/')
        }
      } else if (status === 403) {
        // 403 代表没有权限（我们在后端写的 @RequireRole 或拦截器触发的）
        ElMessage.error('非法操作：您没有权限访问该资源！')
      } else {
        ElMessage.error(error.response.data?.msg || `请求失败 (${status})`)
      }
    } else {
      ElMessage.error('网络连接异常，请检查后端服务是否启动！')
    }
    
    return Promise.reject(error)
  }
)

export default request

// src/utils/ws.js
import { ElMessage } from 'element-plus'

export class WsClient {
  constructor(url) {
    this.url = url
    this.ws = null
    this.heartbeatTimer = null
    this.reconnectTimer = null
    this.isManualClose = false // 是否主动关闭（防止退出时自动重连）
    this.onMessageCallback = null // 收到消息的回调
  }

  // 连接
  connect() {
    this.isManualClose = false
    this.ws = new WebSocket(this.url)

    this.ws.onopen = () => {
      console.log('✅ WebSocket 已连接')
      this.startHeartbeat()
    }

    this.ws.onmessage = (event) => {
      // 忽略心跳回执
      if (event.data === 'PONG') return
      
      // 触发业务回调
      if (this.onMessageCallback) {
        this.onMessageCallback(event.data)
      }
    }

    this.ws.onerror = (e) => {
      console.error('❌ WebSocket 出错', e)
    }

    this.ws.onclose = (e) => {
      console.log('🔌 WebSocket 已断开', e.reason)
      this.stopHeartbeat()
      
      // 如果不是主动关闭，尝试重连
      if (!this.isManualClose) {
        console.log('♻️ 3秒后尝试重连...')
        this.reconnectTimer = setTimeout(() => this.connect(), 3000)
      }
    }
  }

  // 发送消息
  send(msg) {
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(msg)
    }
  }

  // 开启心跳
  startHeartbeat() {
    this.stopHeartbeat()
    this.heartbeatTimer = setInterval(() => {
      this.send('PING')
    }, 30000) // 每 30 秒发一次 PING
  }

  // 停止心跳
  stopHeartbeat() {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer)
      this.heartbeatTimer = null
    }
  }

  // 手动关闭（例如组件卸载或页面退出时）
  close() {
    this.isManualClose = true
    this.stopHeartbeat()
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer)
    if (this.ws) this.ws.close()
  }
}
